import streamlit as st import sqlite3 import hashlib import secrets from datetime import datetime from pathlib import Path # ---------------- PAGE CONFIGURATION ---------------- st.set_page_config( page_title="BlogSpace | Codomax Internship", page_icon="📝", layout="wide" ) DB_FILE = Path(__file__).parent / "blogspace.db" # ---------------- DATABASE ---------------- def get_connection(): conn = sqlite3.connect(DB_FILE) conn.row_factory = sqlite3.Row conn.execute("PRAGMA foreign_keys = ON") return conn def initialize_database(): with get_connection() as conn: conn.executescript(""" CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, email TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, salt TEXT NOT NULL, created_at TEXT NOT NULL ); CREATE TABLE IF NOT EXISTS blogs ( id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT NOT NULL, content TEXT NOT NULL, author_id INTEGER NOT NULL, created_at TEXT NOT NULL, FOREIGN KEY (author_id) REFERENCES users(id) ); """) def hash_password(password, salt=None): if salt is None: salt = secrets.token_hex(16) hashed = hashlib.pbkdf2_hmac( "sha256", password.encode("utf-8"), bytes.fromhex(salt), 200000 ).hex() return hashed, salt def register_user(name, email, password): password_hash, salt = hash_password(password) try: with get_connection() as conn: conn.execute( """ INSERT INTO users (name, email, password_hash, salt, created_at) VALUES (?, ?, ?, ?, ?) """, ( name.strip(), email.strip().lower(), password_hash, salt, datetime.now().strftime("%Y-%m-%d %H:%M:%S") ) ) return True, "Account created successfully!" except sqlite3.IntegrityError: return False, "This email is already registered." def authenticate_user(email, password): with get_connection() as conn: user = conn.execute( "SELECT * FROM users WHERE email = ?", (email.strip().lower(),) ).fetchone() if not user: return None password_hash, _ = hash_password(password, user["salt"]) if secrets.compare_digest(password_hash, user["password_hash"]): return { "id": user["id"], "name": user["name"], "email": user["email"] } return None def create_blog(title, content, author_id): with get_connection() as conn: conn.execute( """ INSERT INTO blogs (title, content, author_id, created_at) VALUES (?, ?, ?, ?) """, ( title.strip(), content.strip(), author_id, datetime.now().strftime("%Y-%m-%d %H:%M:%S") ) ) def get_all_blogs(search=""): with get_connection() as conn: if search.strip(): pattern = f"%{search.strip()}%" return conn.execute( """ SELECT blogs.*, users.name AS author_name FROM blogs JOIN users ON blogs.author_id = users.id WHERE blogs.title LIKE ? OR blogs.content LIKE ? ORDER BY blogs.id DESC """, (pattern, pattern) ).fetchall() return conn.execute( """ SELECT blogs.*, users.name AS author_name FROM blogs JOIN users ON blogs.author_id = users.id ORDER BY blogs.id DESC """ ).fetchall() def get_my_blogs(author_id): with get_connection() as conn: return conn.execute( """ SELECT * FROM blogs WHERE author_id = ? ORDER BY id DESC """, (author_id,) ).fetchall() def get_blog(blog_id): with get_connection() as conn: return conn.execute( """ SELECT blogs.*, users.name AS author_name FROM blogs JOIN users ON blogs.author_id = users.id WHERE blogs.id = ? """, (blog_id,) ).fetchone() def delete_blog(blog_id, author_id): with get_connection() as conn: cursor = conn.execute( "DELETE FROM blogs WHERE id = ? AND author_id = ?", (blog_id, author_id) ) return cursor.rowcount > 0 # ---------------- INITIALIZE ---------------- initialize_database() if "user" not in st.session_state: st.session_state.user = None if "page" not in st.session_state: st.session_state.page = "Home" if "selected_blog" not in st.session_state: st.session_state.selected_blog = None # ---------------- DESIGN ---------------- st.markdown(""" """, unsafe_allow_html=True) st.markdown("""

""", unsafe_allow_html=True) # ---------------- SIDEBAR ---------------- with st.sidebar: st.title("📝 BlogSpace") st.caption("Codomax Internship Project") st.divider() if st.session_state.user: st.success("Welcome, " + st.session_state.user["name"]) page = st.radio( "Navigation", [ "Home", "Create Blog", "All Blogs", "My Blogs" ] ) st.session_state.page = page if st.button("Log Out", use_container_width=True): st.session_state.user = None st.session_state.page = "Home" st.session_state.selected_blog = None st.rerun() else: page = st.radio( "Navigation", [ "Home", "All Blogs", "Login", "Register" ] ) st.session_state.page = page # ---------------- HOME ---------------- if st.session_state.page == "Home": st.subheader("Welcome to BlogSpace! 👋") st.write( "BlogSpace is a simple blog application where users " "can register, log in, publish blog posts, and explore " "content written by other users." ) col1, col2, col3 = st.columns(3) with get_connection() as conn: user_count = conn.execute( "SELECT COUNT(*) FROM users" ).fetchone()[0] blog_count = conn.execute( "SELECT COUNT(*) FROM blogs" ).fetchone()[0] col1.metric("Registered Users", user_count) col2.metric("Published Blogs", blog_count) col3.metric("Database", "SQLite") st.divider() st.subheader("✨ Application Features") a, b = st.columns(2) with a: st.markdown("### 🔐 User Authentication") st.write("Register and log in using a password hash.") st.markdown("### ✍️ Create Blogs") st.write("Write and publish your own blog posts.") with b: st.markdown("### 🔎 Search Blogs") st.write("Find blog posts by title or content.") st.markdown("### 📚 Blog Management") st.write("Read full posts and manage your own blogs.") # ---------------- REGISTER ---------------- elif st.session_state.page == "Register": st.subheader("Create Your Account") with st.form("register_form"): name = st.text_input("Full Name") email = st.text_input("Email Address") password = st.text_input("Password", type="password") confirm = st.text_input("Confirm Password", type="password") submitted = st.form_submit_button( "Register", use_container_width=True ) if submitted: if not name.strip() or not email.strip() or not password: st.warning("Please fill in all fields.") elif "@" not in email or "." not in email.split("@")[-1]: st.warning("Please enter a valid email address.") elif len(password) < 8: st.warning("Password must contain at least 8 characters.") elif password != confirm: st.error("Passwords do not match.") else: success, message = register_user(name, email, password) if success: st.success(message + " Please log in.") else: st.error(message) # ---------------- LOGIN ---------------- elif st.session_state.page == "Login": st.subheader("Login to Your Account") with st.form("login_form"): email = st.text_input("Email Address") password = st.text_input("Password", type="password") submitted = st.form_submit_button( "Login", use_container_width=True ) if submitted: user = authenticate_user(email, password) if user: st.session_state.user = user st.session_state.page = "Home" st.success("Login successful!") st.rerun() else: st.error("Invalid email address or password.") # ---------------- CREATE BLOG ---------------- elif st.session_state.page == "Create Blog": if not st.session_state.user: st.warning("Please log in before creating a blog.") else: st.subheader("✍️ Create a New Blog") with st.form("create_blog_form"): title = st.text_input("Blog Title") content = st.text_area( "Blog Content", height=220, placeholder="Write your thoughts here..." ) submitted = st.form_submit_button( "Publish Blog", use_container_width=True ) if submitted: if not title.strip() or not content.strip(): st.warning("Please enter both title and content.") else: create_blog( title, content, st.session_state.user["id"] ) st.success("Your blog has been published!") st.rerun() # ---------------- ALL BLOGS ---------------- elif st.session_state.page == "All Blogs": st.subheader("📚 Explore Blogs") search = st.text_input("🔎 Search by title or content") blogs = get_all_blogs(search) if not blogs: st.info("No blogs found. Try another search or publish a blog.") for blog in blogs: with st.container(border=True): st.markdown("### " + blog["title"]) st.caption( f"By {blog['author_name']} | {blog['created_at']}" ) preview = blog["content"] if len(preview) > 180: preview = preview[:180] + "..." st.write(preview) if st.button( "Read Full Blog", key=f"read_{blog['id']}" ): st.session_state.selected_blog = blog["id"] st.session_state.page = "Blog Details" st.rerun() # ---------------- MY BLOGS ---------------- elif st.session_state.page == "My Blogs": if not st.session_state.user: st.warning("Please log in to view your blogs.") else: st.subheader("🗂️ My Published Blogs") blogs = get_my_blogs(st.session_state.user["id"]) if not blogs: st.info("You have not published any blogs yet.") for blog in blogs: with st.expander(blog["title"]): st.caption(blog["created_at"]) st.write(blog["content"]) if st.button( "Delete Blog", key=f"delete_{blog['id']}" ): deleted = delete_blog( blog["id"], st.session_state.user["id"] ) if deleted: st.success("Blog deleted successfully.") st.rerun() else: st.error("Could not delete this blog.") # ---------------- INDIVIDUAL BLOG DETAILS ---------------- elif st.session_state.page == "Blog Details": blog_id = st.session_state.selected_blog if blog_id is None: st.warning("Please select a blog first.") else: blog = get_blog(blog_id) if blog is None: st.error("Blog not found.") else: st.subheader(blog["title"]) st.caption( f"Written by {blog['author_name']} " f"on {blog['created_at']}" ) st.divider() st.write(blog["content"]) if st.button("← Back to All Blogs"): st.session_state.page = "All Blogs" st.rerun() # ---------------- FOOTER ---------------- st.divider() st.markdown( "
Developed for Codomax Internship | Module 03
